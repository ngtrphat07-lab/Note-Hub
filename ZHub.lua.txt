-- ============================================================
-- HELPERS

local function track(instance)
    table.insert(ZHub.Instances, instance)
    return instance
end

local function connect(signal, callback)
    local c = signal:Connect(callback)
    table.insert(ZHub.Connections, c)
    return c
end

function UILib:AddTab(name, icon)
    local btn = track(Instance.new("TextButton"))
    btn.Size = UDim2.new(0, 128, 0, 32)
    btn.BackgroundColor3 = ZHub.Theme.Element
    btn.BackgroundTransparency = 0
    btn.Text = "  " .. name
    btn.Font = Enum.Font.GothamSemibold
    btn.TextSize = 13
    btn.TextColor3 = ZHub.Theme.SubText
    btn.TextXAlignment = Enum.TextXAlignment.Left
    btn.AutoButtonColor = false
    btn.Parent = self.sidebar
    local bc = track(Instance.new("UICorner"))
    bc.CornerRadius = UDim.new(0, 6)
    bc.Parent = btn

    local page = track(Instance.new("ScrollingFrame"))
    page.Size = UDim2.new(1, -16, 1, -16)
    page.Position = UDim2.new(0, 8, 0, 8)
    page.BackgroundTransparency = 1
    page.BorderSizePixel = 0
    page.ScrollBarThickness = 3
    page.ScrollBarImageColor3 = ZHub.Theme.Accent
    page.CanvasSize = UDim2.new(0, 0, 0, 0)
    page.AutomaticCanvasSize = Enum.AutomaticSize.Y
    page.Visible = false
    page.Parent = self.content

    local list = track(Instance.new("UIListLayout"))
    list.Parent = page
    list.Padding = UDim.new(0, 6)
    list.SortOrder = Enum.SortOrder.LayoutOrder

    local tab = {button = btn, page = page, name = name}
    table.insert(self.tabs, tab)

    local function select()
        for _, t in ipairs(self.tabs) do
            t.page.Visible = false
            t.button.BackgroundColor3 = ZHub.Theme.Element
            t.button.TextColor3 = ZHub.Theme.SubText
        end
        page.Visible = true
        btn.BackgroundColor3 = ZHub.Theme.Accent
        btn.TextColor3 = ZHub.Theme.Text
        self.currentTab = tab
    end

    btn.MouseButton1Click:Connect(select)
    if #self.tabs == 1 then select() end
    return tab
end

local function makeToggle(page, text, default, callback)
    local flag = text
    ZHub.Flags[flag] = default and true or false

    local btn = track(Instance.new("TextButton"))
    btn.Size = UDim2.new(1, 0, 0, 34)
    btn.BackgroundColor3 = ZHub.Theme.Element
    btn.AutoButtonColor = false
    btn.Text = ""
    btn.Parent = page
    local bc = track(Instance.new("UICorner"))
    bc.CornerRadius = UDim.new(0, 6)
    bc.Parent = btn

    local lbl = track(Instance.new("TextLabel"))
    lbl.Size = UDim2.new(1, -52, 1, 0)
    lbl.Position = UDim2.new(0, 12, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 13
    lbl.TextColor3 = ZHub.Theme.Text
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Text = text
    lbl.Parent = btn

    local sw = track(Instance.new("Frame"))
    sw.Size = UDim2.new(0, 32, 0, 16)
    sw.Position = UDim2.new(1, -40, 0.5, -8)
    sw.BackgroundColor3 = ZHub.Theme.Stroke
    sw.BorderSizePixel = 0
    sw.Parent = btn
    local swC = track(Instance.new("UICorner"))
    swC.CornerRadius = UDim.new(1, 0)
    swC.Parent = sw

    local dot = track(Instance.new("Frame"))
    dot.Size = UDim2.new(0, 12, 0, 12)
    dot.Position = UDim2.new(0, 2, 0.5, -6)
    dot.BackgroundColor3 = ZHub.Theme.Text
    dot.BorderSizePixel = 0
    dot.Parent = sw
    local dC = track(Instance.new("UICorner"))
    dC.CornerRadius = UDim.new(1, 0)
    dC.Parent = dot

    local state = default and true or false
    local function update()
        if state then
            sw.BackgroundColor3 = ZHub.Theme.Accent
            dot.Position = UDim2.new(1, -14, 0.5, -6)
        else
            sw.BackgroundColor3 = ZHub.Theme.Stroke
            dot.Position = UDim2.new(0, 2, 0.5, -6)
        end
        ZHub.Flags[flag] = state
        if callback then callback(state) end
    end
    if default then update() end

    btn.MouseButton1Click:Connect(function()
        state = not state
        update()
    end)

    return {
        Set = function(v) state = v and true or false; update() end,
        Get = function() return state end,
    }
end

-- Element: Slider
local function makeSlider(page, text, minVal, maxVal, default, callback)
    local flag = text
    ZHub.Flags[flag] = default

    local holder = track(Instance.new("Frame"))
    holder.Size = UDim2.new(1, 0, 0, 44)
    holder.BackgroundColor3 = ZHub.Theme.Element
    holder.BorderSizePixel = 0
    holder.Parent = page
    local hc = track(Instance.new("UICorner"))
    hc.CornerRadius = UDim.new(0, 6)
    hc.Parent = holder

    local lbl = track(Instance.new("TextLabel"))
    lbl.Size = UDim2.new(0.6, 0, 0, 20)
    lbl.Position = UDim2.new(0, 12, 0, 6)
    lbl.BackgroundTransparency = 1
    lbl.Font = Enum.Font.Gotham
    lbl.TextSize = 13
    lbl.TextColor3 = ZHub.Theme.Text
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Text = text
    lbl.Parent = holder

    local val = track(Instance.new("TextLabel"))
    val.Size = UDim2.new(0.4, -16, 0, 20)
    val.Position = UDim2.new(0.6, 0, 0, 6)
    val.BackgroundTransparency = 1
    val.Font = Enum.Font.GothamBold
    val.TextSize = 13
    val.TextColor3 = ZHub.Theme.Accent
    val.TextXAlignment = Enum.TextXAlignment.Right
    val.Text = tostring(default)
    val.Parent = holder

    local track_ = track(Instance.new("Frame"))
    track_.Size = UDim2.new(1, -24, 0, 6)
    track_.Position = UDim2.new(0, 12, 0, 30)
    track_.BackgroundColor3 = ZHub.Theme.Stroke
    track_.BorderSizePixel = 0
    track_.Parent = holder
    local tC = track(Instance.new("UICorner"))
    tC.CornerRadius = UDim.new(1, 0)
    tC.Parent = track_

    local fill = track(Instance.new("Frame"))
    fill.Size = UDim2.new((default - minVal) / (maxVal - minVal), 0, 1, 0)
    fill.BackgroundColor3 = ZHub.Theme.Accent
    fill.BorderSizePixel = 0
    fill.Parent = track_
    local fC = track(Instance.new("UICorner"))
    fC.CornerRadius = UDim.new(1, 0)
    fC.Parent = fill

    local dragging = false
    local function setFromX(x)
        local rel = math.clamp((x - track_.AbsolutePosition.X) / track_.AbsoluteSize.X, 0, 1)
        local v = math.floor(minVal + (maxVal - minVal) * rel + 0.5)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        val.Text = tostring(v)
        ZHub.Flags[flag] = v
        if callback then callback(v) end
    end

    track_.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            setFromX(UserInputService:GetMouseLocation().X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            setFromX(input.Position.X)
        end
    end)

    return {
        Set = function(v)
            v = math.clamp(v, minVal, maxVal)
            local rel = (v - minVal) / (maxVal - minVal)
            fill.Size = UDim2.new(rel, 0, 1, 0)
            val.Text = tostring(v)
            ZHub.Flags[flag] = v
            if callback then callback(v) end
        end,
        Get = function() return ZHub.Flags[flag] end,
    }
end

    list.Parent = holder
    local lc = track(Instance.new("UICorner"))
    lc.CornerRadius = UDim.new(0, 6)
    lc.Parent = list

    local ll = track(Instance.new("UIListLayout"))
    ll.Parent = list
    ll.Padding = UDim.new(0, 1)

    local function refresh(newOptions)
        for _, c in ipairs(list:GetChildren()) do
            if c:IsA("TextButton") then c:Destroy() end
        end
        for _, opt in ipairs(newOptions) do
            local b = Instance.new("TextButton")
            b.Size = UDim2.new(1, 0, 0, 25)
            b.BackgroundColor3 = ZHub.Theme.ElementHover
            b.BackgroundTransparency = 0

                val.Text = opt
                list.Visible = false
                if callback then callback(opt) end
            end)
        end
        list.Size = UDim2.new(1, 0, 0, #newOptions * 26)
    end

    refresh(options)

    ddBtn.MouseButton1Click:Connect(function()
        list.Visible = not list.Visible
    end)

    return {
        Refresh = refresh,
        Set = function(opt) val.Text = opt; if callback then callback(opt) end end,
        Get = function() return val.Text end,
    }
end

local flyToggle, flySpeed
local flyConn, flyBP, flyBG
local function startFly(speed)
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end

    if flyBP then flyBP:Destroy() end
    if flyBG then flyBG:Destroy() end
    flyBP = Instance.new("BodyPosition")
    flyBP.Position = root.Position
    flyBP.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    flyBP.Parent = root

    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    flyBG.P = 1e5
    flyBG.CFrame = root.CFrame
    flyBG.Parent = root

    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.RenderStepped:Connect(function()
        if not ZHub.Flags["Fly"] then return end
        local cam = Workspace.CurrentCamera
        local dir = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0,1,0) end
        if dir.Magnitude > 0 then dir = dir.Unit end
        flyBP.Position = root.Position + dir * speed
        flyBG.CFrame = cam.CFrame
    end)
end
local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn = nil end
    if flyBP then flyBP:Destroy(); flyBP = nil end
    if flyBG then flyBG:Destroy(); flyBG = nil end
end

local visTab = UI:AddTab("Visuals")
makeSectionLabel(visTab.page, "ESP Settings")
local espOptions = { Enabled = false, Boxes = true, Names = true, Distance = true, TeamCheck = true }
local espDrawings = {}

local function clearESP()
    for _, d in pairs(espDrawings) do
        for _, inst in ipairs(d) do pcall(function() inst:Remove() end) end
    end
    espDrawings = {}
end

local function makeBoxFor(plr)
    local drawings = {}
    local box = Drawing.new("Quad")
    box.Thickness = 1.5
    box.Color = ZHub.Theme.Accent
    box.Filled = false
    box.Visible = false
    table.insert(drawings, box)

    local name = Drawing.new("Text")
    name.Size = 13
    name.Color = ZHub.Theme.Text
    name.Center = true
    name.Outline = true
    name.Visible = false
    table.insert(drawings, name)

    local dist = Drawing.new("Text")
    dist.Size = 12
    dist.Color = ZHub.Theme.SubText
    dist.Center = true
    dist.Outline = true
    dist.Visible = false
    table.insert(drawings, dist)

    espDrawings[plr] = drawings
end

local function updateESP()
    if not Drawing or not espOptions.Enabled then clearESP(); return end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            if not espDrawings[plr] then makeBoxFor(plr) end
            local char = plr.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local drawings = espDrawings[plr]
            if not root or not hum or hum.Health <= 0 then
                for _, d in ipairs(drawings) do d.Visible = false end
            else
                -- team check
                local sameTeam = false
                if espOptions.TeamCheck and LocalPlayer.Team and plr.Team then
                    sameTeam = LocalPlayer.Team == plr.Team
                end
                if sameTeam then
                    for _, d in ipairs(drawings) do d.Visible = false end
                else
                    local cam = Workspace.CurrentCamera
                    local pos2D, onScreen = cam:WorldToViewportPoint(root.Position)
                    if onScreen then
                        local headPos = cam:WorldToViewportPoint(root.Position + Vector3.new(0, 3, 0))
                        local legPos = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 3, 0))
                        local h = math.abs(headPos.Y - legPos.Y)
                        local w = h / 2
                        local cx, cy = pos2D.X, pos2D.Y
                        drawings[1].PointA = Vector2.new(cx - w/2, cy - h/2)
                        drawings[1].PointB = Vector2.new(cx + w/2, cy - h/2)
                        drawings[1].PointC = Vector2.new(cx + w/2, cy + h/2)
                        drawings[1].PointD = Vector2.new(cx - w/2, cy + h/2)
                        drawings[1].Visible = espOptions.Boxes
                        drawings[2].Position = Vector2.new(cx, cy - h/2 - 14)
                        drawings[2].Text = plr.DisplayName .. " (@" .. plr.Name .. ")"
                        drawings[2].Visible = espOptions.Names
                        local lp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                        local dVal = lp and math.floor((lp.Position - root.Position).Magnitude) or 0
                        drawings[3].Position = Vector2.new(cx, cy + h/2 + 4)
                        drawings[3].Text = tostring(dVal) .. " studs"
                        drawings[3].Visible = espOptions.Distance
                    else
                        for _, d in ipairs(drawings) do d.Visible = false end
                    end
                end
            end
        end
    end
end
